defined_text = {
	name = GetLocalMonth
	random = no

	text = {
		trigger = {
			AND = {
				is_month = 0
				NOT = { is_month = 1 }
			}
		}
		localisation_key = January_Month
	}

	text = {
		trigger = {
			AND = {
				is_month = 1
