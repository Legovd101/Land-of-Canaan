serican_5 = {
	slot = 5
	generic = yes
	ai = yes
	has_country_shield = yes
	potential = { culture_group = greco_serican } 

	seri_deal_with_seleucids = {
		icon = mission_diplomatic_relation
		position = 1
		provinces_to_highlight = {  }

		trigger = {
			OR = {
				is_part_of_hre = yes
				reverse_has_opinion_modifier = {
					who = emperor
					modifier = insulted
				}
			}
		}

		effect = {
			hidden_effect = {
				if = {
					limit = { is_part_of_hre = yes }
					add_adm_power = 50
					add_dip_power = 50
					add_mil_power = 50
					set_country_flag = seri_joined_empire
				}
				else = {
					add_country_modifier = {
						name = military_reform
						duration = 3650
					}
					set_country_flag = seri_insulted_emperor
				}
			}
			custom_tooltip = if_joined_empire
			tooltip = {
				add_adm_power = 50
				add_dip_power = 50
				add_mil_power = 50
			}
			custom_tooltip = if_insulted_emperor
			tooltip = {
				add_country_modifier = {
					name = military_reform
					duration = 3650
				}
			}
		}

	}

	seri_imperial_influence = {
		icon = personal_diplomacy
		position = 2
		required_missions = { seri_deal_with_seleucids } 
		provinces_to_highlight = {
			region = anatolia_region
			OR = {
				AND = {
					ROOT = { has_country_flag = seri_joined_empire }
					is_part_of_hre = yes
				}
				AND = {
					ROOT = { has_country_flag = seri_insulted_emperor }
					is_part_of_hre = no
				}
			}
		}

		trigger = {
			OR = {
				AND = {
					custom_trigger_tooltip = {
						tooltip = if_joined_empire
						has_country_flag = seri_joined_empire
					}
					anatolia_region = {
						type = all
						OR = {
							is_part_of_hre = yes
							owned_by = ROOT
						}
					}
				}
				AND = {
					custom_trigger_tooltip = {
						tooltip = if_insulted_emperor
						has_country_flag = seri_insulted_emperor
					}
					custom_trigger_tooltip = {
						tooltip = anatolia_not_in_empire
						anatolia_region = {
							type = all
							is_part_of_hre = no
						}
					}
				}
			}
		}

		effect = {
			hidden_effect = {
				if = {
					limit = { has_country_flag = seri_joined_empire }
					add_casus_belli = {
						type = cb_claim_emperorship
						target = emperor
						months = 600
					}
				}
				if = {
					limit = { has_country_flag = seri_insulted_emperor }
					emperor = {
						every_owned_province = { add_permanent_claim = ROOT }
					}
				}
			}
			custom_tooltip = if_joined_empire
			tooltip = {
				add_casus_belli = {
					type = cb_claim_emperorship
					target = emperor
					months = 600
				}
			}
			custom_tooltip = if_insulted_emperor
			tooltip = {
				emperor = {
					every_owned_province = { add_permanent_claim = ROOT }
				}
			}
		}
	}

	seri_masters_of_asia = {
		icon = polish_diplomacy
		position = 3
		required_missions = { seri_imperial_influence } 
		provinces_to_highlight = {
			is_capital_of = emperor
		}

		trigger = {
			OR = {
				AND = {
					custom_trigger_tooltip = {
						tooltip = if_joined_empire

					}
					is_emperor = yes
				}
				AND = {
					custom_trigger_tooltip = {
						tooltip = if_insulted_emperor
						has_country_flag = seri_insulted_emperor
					}
					emperor = {
						capital_scope = {
							area_for_scope_province = {
								type = all
								controlled_by = ROOT
							}
						}
					}
				}
			}
		}

		effect = {
			hidden_effect = {
				if = {
					limit = { has_country_flag = seri_joined_empire }
					add_authority = 40
				}
				if = {
					limit = { has_country_flag = seri_insulted_emperor }
					mashriq_region = {
						add_permanent_claim = ROOT
					}
					dismantle_hre = yes
				}
			}
			custom_tooltip = if_joined_empire
			tooltip = {
				add_authority = 40
			}
			custom_tooltip = if_insulted_emperor
			tooltip = {
				mashriq_region = {
					add_permanent_claim = ROOT
				}
				dismantle_hre = yes
			}

	}
