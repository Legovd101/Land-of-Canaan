namespace = flavor_GDL

# The War of the Gaelic Coalition (Gaedlighe Intro Event)
country_event = {
	id = flavor_GDL.1
	title = flavor_GDL.1.title
	picture = SIEGE_PRAGUE_CASTLE_eventPicture
	desc = {
		trigger = { tag = GDL }
		desc = flavor_GDL.1.desc_GDL
	}
	desc = {
		trigger = { tag = ALN }
		desc = flavor_GDL.1.desc_ALN
	}
	desc = {
		trigger = { tag = KSB }
		desc = flavor_GDL.1.desc_KSB
	}

	is_triggered_only = yes
	trigger = {
		OR = {
			tag = GDL
			tag = ALN
			tag = KSB
		}
	}
	
	immediate = { set_country_flag = in_gaelic_war }

	option = { # We shall triumph!
		name = flavor_GDL.1.a
		if = {
			custom_tooltip = GDL_intro_tt
			limit = { tag = GDL}
			while = {
				limit = { NOT = { army_size_percentage = 1.0 } }
				capital_scope = { infantry = GDL }
			}
		}
		else_if = {
			custom_tooltip = ALN_intro_tt
			limit = { tag = ALN }
			create_general = {
				tradition = 75
				name = "Bagawas"
			}
		}
		else_if = {
			custom_tooltip = KSB_intro_tt
			limit = { tag = KSB }
			create_general = {
				tradition = 75
				name = "Cixilsun"
			}
		}
	}
}

# An Offer of Peace? (Gaedlighe Losing)
country_event = {
	id = flavor_GDL.2
	title = flavor_GDL.2.title
	picture = GOOD_WITH_MONARCH_eventPicture
	desc = flavor_GDL.2.desc
	
	trigger = {
		tag = ALN
		war_score_against = {
			who = GDL
			value = 50
		}
		has_country_flag = in_gaelic_war
	}
	mean_time_to_happen = {
		days = 1
	}
	fire_only_once = yes
		
	option = { # Let us send them an ultimatum!
		name = flavor_GDL.2.a
		custom_tooltip = treaty_of_asturias_lose_tt
		GDL = {
			country_event = {
				id = flavor_GDL.3
				days = 3
			}
		}
	}

	option = { # I can handle my own peace treaty!
		name = flavor_GDL.2.b
		ai_chance = {
			factor = 0
		}
	}
}

# Consolidation of the Suebic Counts
country_event = {
	id = flavor_GDL.6
	title = flavor_GDL.6.title
	picture = GOOD_WITH_MONARCH_eventPicture
	desc = flavor_GDL.6.desc
	
	trigger = {
		tag = GDL
		any_subject_country = {
			is_subject_of_type = tributary_state
			culture = suebi
			has_opinion = {
				who = ROOT
				value = 150
			}
			NOT = { liberty_desire = 50 }
			alliance_with = ROOT
			ai = yes
		}
	}
	
	mean_time_to_happen = {
		months = 12
	}

	option = { # Perfect!
		name = flavor_GDL.6.a
		random_country = {
			limit = {
				is_subject_of_type = tributary_state
				culture = suebi
				has_opinion = {
					who = ROOT
					value = 150
				}
				NOT = { liberty_desire = 50 }
				alliance_with = ROOT
				ai = yes
			}
			ROOT = { vassalize = PREV }
		}
	}
}
