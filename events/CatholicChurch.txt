# The Koiné Akadimía for the emperor...
country_event = {
	id = itloc_catholic_events.1
	title = "itloc_catholic_events.1.title"
	desc = "itloc_catholic_events.1.desc"
	picture = HRE_eventPicture
	
	major = yes
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			set_country_flag = common_academy_flag
		}
	}

	option = { # These "Katholikon" shall lead the empire into the future!
		name = "itloc_catholic_events.1.a"
		change_religion = katholikos
		capital_scope = {
			change_religion = katholikos
		}
		set_hre_religion = katholikos
		every_country = {
			limit = {
				is_part_of_hre = yes
				is_emperor = no
				num_of_cities = 1
				NOT = { vassal_of = ROOT }
			}
			country_event = { id = itloc_catholic_events.2 }
		}
	}
	option = { # This all sounds far too radical...
		name = "itloc_catholic_events.1.b"
	}
}
