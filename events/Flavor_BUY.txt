# Disagreement between the Kas!
country_event = {
	id = flavor_BUY.8
	title = flavor_BUY.8.title
	picture = shinto_DIPLOMACY_eventPicture
	desc = flavor_BUY.8.desc
	
	trigger = {
		tag = BUY
		any_owned_province = {
			any_neighbor_province = {
				region = bactria_region
				NOT = { owned_by = ROOT }
			}
		}
	}
	mean_time_to_happen = {
		years = 10
	}
	fire_only_once = yes
		
	option = { # Oh, must we squabble? 
		name = flavor_BUY.8.a
		random = {
			chance = 25
			add_stability = -1
		}
	}
		
	option = { # ...Why are we even arguing about this?
		name = flavor_BUY.8.b
		ai_chance = {
			factor = 0
		}
		add_dip_power = -100
		random_province = {
			limit = {
				NOT = { owned_by = ROOT }
				any_neighbor_province = { owned_by = ROOT }
				region = bactria_region
			}
			area = { add_claim = ROOT }
		}
	}
}
