# Deciding on the path of the Heavenly Kingdom
country_event = {
	id = itloc_heavenly.4
	title = itloc_heavenly.4.title
	desc = itloc_heavenly.4.desc
	picture = ICONS_IN_CHURCH_eventPicture
	
	is_triggered_only = yes

	trigger = { is_in_heavenly_alliance = yes }

	option = { # The teachings of Confucius serve us well enough.
		name = itloc_heavenly.4.a

		hidden_effect = {
			every_subject_country = {
				limit = { is_in_heavenly_alliance = yes }
				change_religion = ROOT
				set_ruler_religion = ROOT
				set_consort_religion = ROOT
				set_heir_religion = ROOT
			}
		}

	}

	option = { # We shall follow the teachings of the Buddha.
		name = itloc_heavenly.4.b

		change_religion = mahayana
		capital_scope = { change_religion = mahayana }
		hidden_effect = {
			set_ruler_religion = mahayana
			set_consort_religion = mahayana
			set_heir_religion = mahayana
		}
		hidden_effect = {
			every_subject_country = {
				limit = { is_in_heavenly_alliance = yes }
				change_religion = ROOT
				set_ruler_religion = ROOT
				set_consort_religion = ROOT
				set_heir_religion = ROOT
			}
		}

	}

	option = { # Henceforth, the state shall enforce the teachings of Laozi.
		name = itloc_heavenly.4.c

		change_religion = daoism
		capital_scope = { change_religion = daoism }
		hidden_effect = {
			set_ruler_religion = daoism
			set_consort_religion = daoism
			set_heir_religion = daoism
		}
		hidden_effect = {
			every_subject_country = {
				limit = { is_in_heavenly_alliance = yes }
				change_religion = ROOT
				set_ruler_religion = ROOT
				set_consort_religion = ROOT
				set_heir_religion = ROOT
			}
		}

	}

}
