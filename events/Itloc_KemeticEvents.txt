# The Western Nome
country_event = {
	id = itloc_kemetic.1
	title = itloc_kemetic.1.title
	desc = itloc_kemetic.1.desc
	picture = COURT_eventPicture

	is_triggered_only = yes
	
	option = {
		name = itloc_kemetic.1.a
		every_owned_province = {
			limit = {
				NOT = { area = cyrenaica_area }
			}
			remove_core = CYR
		}
		release = CYR
		create_march = CYR
	}
	
	option = {
		name = itloc_kemetic.1.b
		cyrenaica_area = {
			add_core = ROOT
		}
	}
}
