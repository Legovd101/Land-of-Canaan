namespace = flavor_ROM

# The Fall of Rome
country_event = {
	id = flavor_ROM.1
	title = flavor_ROM.1.title
	picture = SIEGE_PRAGUE_CASTLE_eventPicture
	desc = {
		trigger = {
			NOT = {
				any_army = {
					is_ruler_commanding_unit = yes
				}
			}
		}
		desc = flavor_ROM.1.desc.a
	}
	desc = {
		trigger = {
			any_army = {
				is_ruler_commanding_unit = yes
			}
		}
		desc = flavor_ROM.1.desc.b
	}

	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		tag = ROM
	}
	
	immediate = { set_country_flag = invading_bactria }

	option = { # Aaaaaaah!
		name = flavor_ROM.1.a
		trigger = {
			NOT = {
				any_army = {
					is_ruler_commanding_unit = yes
				}
			}
		}
		random = {
			chance = 99
			kill_ruler = yes
		}
		add_prestige = -50
	}

	option = { # No... No!
		name = flavor_ROM.1.b
		trigger = {
			any_army = {
				is_ruler_commanding_unit = yes
			}
		}
		add_prestige = -50
	}
}
