# Lietuvan Nobles Elect their Own King!

	id = flavor_PRU.2
	title = flavor_PRU.2.title
	picture = ADVISOR_eventPicture
	desc = flavor_PRU.2.desc
	
	fire_only_once = yes
	
	major = yes
	
	trigger = {
		tag = PRU
		NOT = { has_country_modifier = philosopher_king }
		NOT = { has_country_flag = lithuanian_revolt }
		LIT = { junior_union_with = PRU }
		NOT = { is_year = 1500 }
	}
	
	mean_time_to_happen = {
		months = 48
	}
	
	immediate = {
		set_country_flag = lithuanian_revolt
	}
	
	option = { # I couldn't care less, let them go for all I care! Just leave me be!
		name = "flavor_PRU.2.a"
		ai_chance = {
			factor = 20
		}
		LIT = {
			define_ruler = {
				name = "Netimari"
				dynasty = "Deltuviai"
				adm = 4
				dip = 2
				mil = 6
				claim = 50
			}
			break_union = PRU
		}
		add_prestige = -25
	}
	
	option = { # Wait, they *what* now? They can't do that, send in the army!
		name = "flavor_PRU.2.b"
		ai_chance = {
			factor = 1
		}
		LIT = {

				claim = 50
			}

		}
	}
