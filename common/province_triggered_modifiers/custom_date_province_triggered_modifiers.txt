day_tick_modifier = {

	potential = {
		always = yes
	}

	trigger = {
		NOT = { has_province_flag = day_tick_flag }
	}
	
	hidden = yes
	
	on_activation = {
		day_tick = yes
		set_province_flag = day_tick_flag
	}
	
	on_deactivation = {
		day_tick = yes
		clr_province_flag = day_tick_flag
	} 
}
