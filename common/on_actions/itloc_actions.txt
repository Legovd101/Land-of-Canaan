on_startup = {
	hre_inheritable = yes
	if = {
		limit = {
			tag = XIM
		}
		random_country = {
			limit = {
				culture_group = greco_serican
				num_of_cities = 3
			}
			country_event = {
				id = itloc_serican.2
			}
		}
	}
	events = {
		flavor_GDL.1
		flavor_BUY.1
		# marcin_events.103
		itloc_serican.1
	}
	1 = {
		province_event = { id = prices_marcin.13 days = 365 }
	}

}

on_war_won = {
	if = {
		limit = { has_country_flag = in_gaelic_war }
		if = {
			limit = { FROM = { tag = ALN } }
			clr_country_flag = in_gaelic_war
			set_country_flag = won_gaelic_war
			FROM = { set_country_flag = lost_gaelic_war clr_country_flag = in_gaelic_war }
		}
		else = {
			FROM = { set_country_flag = lost_gaelic_war clr_country_flag = in_gaelic_war }
		}
	}
	if = {
		limit = { has_country_flag = invading_bactria }
		if = {
			limit = { FROM = { tag = MKN } }
			clr_country_flag = invading_bactria
			set_country_flag = invaded_bactria_successfully
		}
	}
}
on_war_lost = {
	if = {
		limit = { has_country_flag = invading_bactria }
		if = {
			limit = { FROM = { tag = MKN } }
			clr_country_flag = invading_bactria
			set_country_flag = didnt_invade_bactria
		}
	}
}

on_siege_won_province = {
	if = {
		limit = {
			controller = { has_country_flag = invading_bactria }
			province_id = 454
		}
		controller = {
			country_event = {
				id = flavor_BUY.7
			}
		}
	}
	if = {
		limit = {
			province_id = 118
		}
		controller = {
			country_event = {
				id = flavor_ROM.2
			}
		}
		owner = {
			country_event = {
				id = flavor_ROM.1
			}
		}
	}
}

on_core = {
	if = {
		limit = {
			owner = { is_part_of_hre = yes }
			# NOT = { religion_group = emperor }
			OR = {
				any_neighbor_province = { is_part_of_hre = yes }
				any_friendly_coast_border_province  = { is_part_of_hre = yes }
			}
			is_part_of_hre = no
		}
		set_in_empire = yes
	}
}
