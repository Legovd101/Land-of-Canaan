# Latin league Effects
league_successor_calculation_effect = {
	save_global_event_target_as = current_successor
	while = {
		limit = {
			any_subject_country = {
				is_subject_of_type = saangfan
				check_variable = {
					which = heavenly_succession_score
					which = event_target:current_successor
				}
			}
		}
		random_subject_country = {
			limit = {
				is_subject_of_type = saangfan
				check_variable = {
					which = heavenly_succession_score
					which = event_target:current_successor
				}
			}
			save_global_event_target_as = current_successor
		}
	}
}

league_unity_calculation_effect = {
	set_variable = {
		which = tungjat_gain
		value = 0
	}
	set_variable = {
		which = heavenly_provinces_count
		value = 0
	}
	set_variable = {
		which = heavenly_heretic_provinces
		value = 0
	}
	set_variable = {
		which = heavenly_alliance_members
		value = 0
	}
	set_variable = {
		which = heavenly_member_gain
		value = 0
	}
	set_variable = {
		which = heavenly_internal_war_members
		value = 0
	}
	set_variable = {
		which = heavenly_defending_members
		value = 0
	}
	set_variable = {
		which = heavenly_attacking_members
		value = 0
	}
	set_variable = {
		which = heavenly_internal_war_gain
		value = 0
	}
	set_variable = {
		which = heavenly_defending_gain
		value = 0
	}
	set_variable = {
		which = heavenly_attacking_gain
		value = 0
	}
	set_variable = {
		which = heavenly_peacetime_gain
		value = 0
	}

	## League Religious Unity Calculation
	every_province = {
		limit = { owner = { is_in_heavenly_alliance = yes } }
		event_target:current_tinzi = {
			change_variable = {
				which = heavenly_provinces_count
				value = 1
			}
		}
		if = {
			limit = { NOT = { religion = ROOT } }
			event_target:current_tinzi = {
				change_variable = {
					which = heavenly_heretic_provinces
					value = 1
				}
			}
		}
	}
	set_variable = {
		which = heavenly_religious_unity
		which = heavenly_heretic_provinces
	}
	divide_variable = {
		which = heavenly_religious_unity
		which = heavenly_provinces_count
	}
	set_variable = {
		which = heavenly_religious_unity_gain
		which = heavenly_religious_unity
	}
	multiply_variable = {
		which = heavenly_religious_unity_gain
		value = 100
	}
	round_variable = {
		which = heavenly_religious_unity_gain
		value = 0
	}
	divide_variable = {
		which = heavenly_religious_unity_gain
		value = 100
	}
	subtract_variable = {
		which = heavenly_religious_unity_gain
		value = 0.5
	}
	multiply_variable = {
		which = heavenly_religious_unity_gain
		value = -1
	}
	multiply_variable = {
		which = heavenly_religious_unity
		value = -100
	}
	change_variable = {
		which = heavenly_religious_unity
		value = 100
	}
	round_variable = {
		which = heavenly_religious_unity
		value = 0
	}

	## League Member Calculations
	every_country = {
		limit = { is_in_heavenly_alliance = yes }
		event_target:current_tinzi = {
			change_variable = {
				which = heavenly_alliance_members
				value = 1
			}	
			change_variable = {
				which = heavenly_member_gain
				value = 0.1
			}
		}
		### War Calculations
		#### Internal War Debuff
		if = {
			limit = {
				any_country = {
					is_in_heavenly_alliance = yes
					OR = {
						is_in_war = {
							attacker_leader = PREV
							defender_leader = THIS
						}
						is_in_war = {
							attacker_leader = THIS
							defender_leader = PREV
						}
					}
				}
			}
			event_target:current_tinzi = {
				change_variable = {
					which = heavenly_internal_war_gain
					value = 0.5
				}
				change_variable = {
					which = heavenly_internal_war_members
					value = 1
				}
			}
		}
		#### Offensive War Buff
		if = {
			limit = {
				OR = {
					any_country = {
						is_in_heavenly_alliance = no
						is_in_war = {
							attacker_leader = PREV
							defender_leader = THIS
						}
					}
					any_country = {
						is_in_heavenly_alliance = yes
						any_country = {
							is_in_heavenly_alliance = no
							is_in_war = {
								attacker_leader = PREV
								defender_leader = THIS
							}
						}
						PREV = {
							offensive_war_with = PREV
						}
					}
				}
			}
			event_target:current_tinzi = {
				change_variable = {
					which = heavenly_attacking_gain
					value = 0.2
				}
				change_variable = {
					which = heavenly_attacking_members
					value = 1
				}
			}
		}
		#### Defensive War Buff
		if = {
			limit = {
				OR = {
					any_country = {
						is_in_heavenly_alliance = no
						is_in_war = {
							attacker_leader = THIS
							defender_leader = PREV
						}
					}
					any_country = {
						is_in_heavenly_alliance = yes
						any_country = {
							is_in_heavenly_alliance = no
							is_in_war = {
								attacker_leader = THIS
								defender_leader = PREV
							}
						}
						PREV = {
							defensive_war_with = PREV
						}
					}
				}
			}
			event_target:current_tinzi = {
				change_variable = {
					which = heavenly_defending_gain
					value = 0.25
				}
				change_variable = {
					which = heavenly_defending_members
					value = 1
				}
			}
		}
	}
	if = {
		limit = {
			all_country = {
				OR = {
					is_in_heavenly_alliance = no
					is_at_war = no
				}
			}
		}
		event_target:current_tinzi = {
			change_variable = {
				which = heavenly_peacetime_gain
				value = 1.0
			}
		}						
	}
	## Final Unity Calculation
	if = {
		limit = { NOT = { has_global_flag = league_religion_abolished } }
		change_variable = {
			which = tungjat_gain
			which = heavenly_religious_unity_gain
		}
	}

	subtract_variable = {
		which = tungjat_gain
		which = heavenly_internal_war_gain
	}

	change_variable = {
		which = tungjat_gain
		which = heavenly_member_gain
	}

	change_variable = {
		which = tungjat_gain
		which = heavenly_defending_gain
	}

	change_variable = {
		which = tungjat_gain
		which = heavenly_attacking_gain
	}

	change_variable = {
		which = tungjat_gain
		which = heavenly_peacetime_gain
	}

	change_variable = {
		which = tungjat
		which = tungjat_gain
	}
	
	if = {
		limit = {
			check_variable = {
				which = tungjat
				value = 100
			}
		}
		set_variable = {
			which = tungjat
			value = 100
		}
	}
}

league_succession_score_effect = {
	## Base Change/Ruler Dip Change
	export_to_variable = {
		which = heavenly_ruler_succession
		value = trigger_value:monthly_dip
	}
	change_variable = {
		which = heavenly_succession_score
		value = 1
	}
	change_variable = {
		which = heavenly_succession_score
		which = heavenly_ruler_succession
	}
	## Has League Religion, 50% League Religious Unity, 60% Individual Religious Unity
	if = {
		limit = {
			OR = {
				overlord = {
					check_variable = {
						which = heavenly_religious_unity
						value = 50
					}
				}
				check_variable = {
					which = heavenly_religious_unity
					value = 50
				}
			}
			NOT = { has_global_flag = league_religion_abolished }
			OR = {
				has_subject_of_type = saangfan
				religion = overlord
			}
			religious_unity = 0.60
		}
		change_variable = {
			which = heavenly_succession_score
			value = 5
		}
	}

}
